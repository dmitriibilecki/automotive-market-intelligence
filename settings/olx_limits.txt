# Главные настройки ручного запуска main.py.
# Меняешь цифры тут, сохраняешь файл и запускаешь:
# .\.venv\Scripts\python.exe .\main.py
#
# Формат простой: ключ=значение
# Строки с # не читаются.

# Что запускать:
# olx_listings    - новый сбор объявлений в SQLite + Excel
# brand_snapshot  - только количество марок Польша/Варшава
# warsaw_details  - старый отдельный Excel по Варшаве
# all             - все режимы подряд, обычно не надо
mode=olx_listings

# По умолчанию работаем по Варшаве, а не по всей Польше.
olx_url=https://www.olx.pl/motoryzacja/samochody/warszawa/

# Сколько страниц поиска открыть.
# Для проверки ставь 1.
# Для 150 объявлений нужно примерно 5 страниц, потому что одна страница OLX дает около 30-36 карточек.
listings_max_pages=15

# Сколько объявлений максимум обработать за запуск.
# Сейчас цель: до 150 объявлений за запуск/сутки.
listings_max_listings=150

# false = браузер видно глазами.
# true  = браузер работает в фоне.
headless=true

# Пауза между карточками объявлений в секундах.
# Чем больше пауза, тем спокойнее и медленнее сбор.
min_delay=5
max_delay=15

# Настройки для старого режима warsaw_details.
# Они нужны только если запускаешь mode=warsaw_details.
warsaw_url=https://www.olx.pl/motoryzacja/samochody/warszawa/
max_pages=1
max_listings=10
